33async def _call_llm(prompt: str, system: str, model: str, temperature: float = 0.2, max_tokens: int = 1024) -> tuple[str, str | None]: 

34 """Call the LLM gateway for actual inference. Returns (text, error).""" 

35 try: 

36 async with httpx.AsyncClient(timeout=httpx.Timeout(120, connect=10)) as client: 

37 response = await client.post( 

38 f"{LLM_GATEWAY_URL}/v1/generate", 

39 headers=outgoing_headers(), 

40 json={ 

41 "prompt": prompt, 

42 "system": system, 

43 "model": model, 

44 "temperature": temperature, 

45 "max_tokens": max_tokens, 

46 }, 

47 ) 

48 if response.status_code < 400: 

49 return response.json().get("text", "").strip(), None 

50 return "", f"LLM gateway returned HTTP {response.status_code}" 

51 except Exception as exc:

55def run_agent(req: AgentRunRequest) -> dict: 

56 """ 

57 Execute an agent with real LLM inference. 

58 

59 FIX: Previously returned a hardcoded Arabic string. Now: 

60 1. Routes to the correct model based on task sensitivity 

61 2. Calls the LLM gateway for actual inference 

62 3. Falls back gracefully if LLM is unavailable 

63 """ 

64 import asyncio 

65 import logging 

66 started = time.time() 

67 run_id = f"run_{uuid.uuid4().hex[:12]}" 

68 agent = AGENTS.get(req.agent_id, AGENTS["supervisor"]) 

69 sensitivity = "restricted" if agent.get("risk") == "restricted" else ("high" if agent.get("risk") == "high" else "medium") 

70 model_route = route_model(type("RouteReq", (), {"task": req.task, "sensitivity": sensitivity, "language": "ar", "max_latency_ms": None, "require_local_only": True})()) 

71 selected_tools = req.tools or agent["tools"] 

72 

73 # FIX: Actually call the LLM instead of returning static text 

74 system_prompt = ( 

75 f"You are {agent['name']} in the HSAAI enterprise platform. " 

76 f"Respond in Arabic. Be concise and professional. " 

77 f"Available tools: {', '.join(selected_tools)}." 

78 ) 

79 # FIX-15: _call_llm is async — must run via event loop. Previously 

80 # called without await, producing "coroutine was never awaited" warning 

81 # and never actually invoking the LLM. 

82 try: 

83 try: 

84 asyncio.get_running_loop() 

85 # Already in an event loop — schedule a new one in a thread to avoid 

86 # "cannot run from a running event loop" errors when called from async 

87 # routers. Use asyncio.run in a fresh loop via thread. 

88 import concurrent.futures 

89 from contextvars import copy_context 

90 context = copy_context() 

91 with concurrent.futures.ThreadPoolExecutor(max_workers=1) as ex: 

92 answer, llm_error = ex.submit(context.run, lambda: asyncio.run(_call_llm(req.task, system_prompt, model_route["model"]))).result() 

93 except RuntimeError: 

94 # No running loop — safe to use asyncio.run directly 

95 answer, llm_error = asyncio.run(_call_llm(req.task, system_prompt, model_route["model"])) 

96 except Exception as exc: 

97 logging.getLogger("hsaai.phase5.agent_runtime").warning( 

98 "LLM call failed (non-fatal): %s", exc 

99 ) 

100 answer, llm_error = "", str(exc) 

101 

102 if not answer and not llm_error: 

103 llm_error = "Model returned an empty response" 

104 succeeded = bool(answer) and llm_error is None 

105 

106 result = { 

107 "run_id": run_id, 

108 "agent_id": req.agent_id, 

109 "agent_name": agent["name"], 

110 "status": "completed" if succeeded else "failed", 

111 "task": req.task, 

112 "selected_tools": selected_tools, 

113 "model_route": model_route, 

114 "execution_trace": [ 

115 {"step": "policy_check", "status": "passed", "detail": "tenant/workspace scoped execution"}, 

116 {"step": "tool_selection", "status": "completed", "tools": selected_tools}, 

117 {"step": "model_route", "status": "completed", "model": model_route["model"]}, 

118 {"step": "llm_inference", "status": "completed" if succeeded else "failed", "error": llm_error}, 

119 {"step": "response_contract", "status": "completed" if succeeded else "failed", "sources_required": req.require_sources}, 

120 ], 

121 "answer": answer, 

122 "llm_error": llm_error, 

123 "elapsed_ms": int((time.time()-started)*1000), 

124 } 

125 record_event(ObservabilityEvent(event_type="agent_run", component="agent_runtime", tenant_id=req.context.tenant_id, workspace_id=req.context.workspace_id, latency_ms=result["elapsed_ms"], model=model_route["model"], success=succeeded, risk_level=agent.get("risk", "medium"), metadata={"agent_id": req.agent_id, "run_id": run_id, "llm_used": not bool(llm_error)})) 

126 return result
