79def run_migrations() -> None: 

80 """Run database migrations. 

81 

82 FIX D-06: USE_ALEMBIC defaults to 'true' now. Production/staging refuse 

83 to run with USE_ALEMBIC=false (RLS, indexes, constraints would be missing). 

84 """ 

85 # FIX D-06: default to 'true' (was 'false') 

86 use_alembic = os.getenv("USE_ALEMBIC", "true").lower() == "true" 

87 app_env = os.getenv("APP_ENV", "development").lower() 

88 

89 if not use_alembic and app_env in {"production", "prod", "staging"}: 

90 # Refuse to run without alembic in production 

91 raise RuntimeError( 

92 "USE_ALEMBIC=false is forbidden in production/staging. " 

93 "Set USE_ALEMBIC=true and ensure migrations are tested." 

94 ) 

95 

96 if os.getenv("RUN_MIGRATIONS_ON_STARTUP", "true").lower() == "false": 

97 if not use_alembic: 

98 raise RuntimeError("Schema verification requires USE_ALEMBIC=true") 

99 from alembic.config import Config 

100 from alembic.script import ScriptDirectory 

101 root = Path(__file__).resolve().parents[3] 

102 cfg = Config(str(root / "alembic.ini")) 

103 cfg.set_main_option("script_location", str(root / "alembic")) 

104 expected = ScriptDirectory.from_config(cfg).get_current_head() 

105 with engine.connect() as connection: 

106 revisions = set(connection.execute(text("SELECT version_num FROM alembic_version")).scalars()) 

107 if revisions != {expected}: 

108 raise RuntimeError("Database schema is not current; run the migration job") 

109 if connection.dialect.name == "postgresql": 

110 unsafe = connection.execute(text("SELECT rolsuper OR rolbypassrls FROM pg_roles WHERE rolname = current_user")).scalar() 

111 if unsafe: 

112 raise RuntimeError("Runtime database role must not bypass RLS") 

113 return 

114 

115 if use_alembic: 

116 logger.info("USE_ALEMBIC=true — running `alembic upgrade head`") 

117 candidates = [ 

118 Path.cwd() / "alembic.ini", 

119 Path(__file__).resolve().parent.parent.parent.parent / "alembic.ini", 

120 Path(__file__).resolve().parent.parent.parent / "alembic.ini", 

121 ] 

122 alembic_ini = next((p for p in candidates if p.exists()), None) 

123 if alembic_ini is None: 

124 raise RuntimeError("USE_ALEMBIC=true but alembic.ini is missing; refusing schema fallback") 

125 try: 

126 result = subprocess.run( 

127 [sys.executable, "-m", "alembic", "-c", str(alembic_ini), "upgrade", "head"], 

128 check=True, 

129 capture_output=True, 

130 text=True, 

131 timeout=600, 

132 ) 

133 logger.info("Alembic migration completed:\n%s", result.stdout[-2000:]) 

134 if result.stderr: 

135 logger.warning("Alembic stderr:\n%s", result.stderr[-1000:]) 

136 except subprocess.CalledProcessError as exc: 

137 logger.error("Alembic migration FAILED (exit %d):\n%s\n%s", 

138 exc.returncode, exc.stdout[-1000:], exc.stderr[-1000:]) 

139 raise 

140 except FileNotFoundError: 

141 logger.error("`alembic` command not found. Install with: pip install alembic") 

142 raise 

143 else: 

144 logger.warning( 

145 "run_migrations() is using Base.metadata.create_all() — dev only. " 

146 "For production, set USE_ALEMBIC=true and run `alembic upgrade head`."
