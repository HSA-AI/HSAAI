16 def templates(self) -> dict[str, Any]:

19 def start(self, db: Session, payload: WorkflowStartRequest) -> dict[str, Any]: 

20 if payload.template_key not in WORKFLOW_TEMPLATES: 

21 raise ValueError("Unsupported workflow template") 

22 template = WORKFLOW_TEMPLATES[payload.template_key] 

23 execution_id = f"wf-{payload.template_key}-{int(time.time()*1000)}" 

24 row = WorkflowExecution(execution_id=execution_id, template_key=payload.template_key, title=payload.title, requested_by=payload.requested_by, payload_json=json.dumps(payload.payload, ensure_ascii=False), steps_json=json.dumps(template["steps"], ensure_ascii=False), current_step=template["steps"][0], tenant_id=payload.tenant_id, workspace_id=payload.workspace_id) 

25 db.add(row); db.flush() 

26 db.add(WorkflowAuditEvent(execution_id=execution_id, actor=payload.requested_by, action="start", comment=f"Started {template['name']}", status_after="running")) 

27 db.commit() 

28 return self.serialize(row) | {"template": template}

30 def action(self, db: Session, payload: WorkflowActionRequest) -> dict[str, Any]: 

31 row = db.query(WorkflowExecution).filter_by(execution_id=payload.execution_id).first() 

32 if not row: raise ValueError("Workflow execution not found") 

33 steps = json.loads(row.steps_json or "[]"); idx = steps.index(row.current_step) if row.current_step in steps else 0 

34 if payload.action == "reject": row.status = "rejected" 

35 elif payload.action == "escalate": row.status = "escalated"; row.sla_status = "at_risk" 

36 elif idx >= len(steps) - 1: row.status = "completed"; row.current_step = steps[-1] 

37 else: row.current_step = steps[idx + 1]; row.status = "running" 

38 db.add(WorkflowAuditEvent(execution_id=row.execution_id, actor=payload.actor, action=payload.action, comment=payload.comment, status_after=row.status)) 

39 db.commit(); return self.serialize(row)

41 def list(self, db: Session, tenant_id: str, workspace_id: str) -> dict[str, Any]: 

42 rows = db.query(WorkflowExecution).filter_by(tenant_id=tenant_id, workspace_id=workspace_id).order_by(WorkflowExecution.id.desc()).limit(100).all() 

43 return {"executions": [self.serialize(r) for r in rows], "summary": {"running": sum(1 for r in rows if r.status == "running"), "completed": sum(1 for r in rows if r.status == "completed"), "escalated": sum(1 for r in rows if r.status == "escalated")}}
