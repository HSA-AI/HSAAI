83 @field_validator("keycloak_client_secret", "smtp_password") 

84 @classmethod 

85 def reject_placeholder_secret(cls, value: str | None): 

86 if value and value.lower() in {"change-me", "change-me-in-production", "secret", "password"}:

90 @model_validator(mode="after") 

91 def validate_production(self): 

92 if self.is_production: 

93 if self.effective_database_url.startswith("sqlite"): 

94 raise ValueError("SQLite is forbidden in production. Set DATABASE_URL to PostgreSQL.") 

95 if "postgresql" not in self.effective_database_url: 

96 raise ValueError("Production DATABASE_URL must be PostgreSQL.") 

97 required = { 

98 "KEYCLOAK_URL": self.keycloak_url, 

99 "KEYCLOAK_REALM": self.keycloak_realm, 

100 "KEYCLOAK_CLIENT_ID": self.keycloak_client_id, 

101 "QDRANT_URL": self.qdrant_url, 

102 "QDRANT_COLLECTION": self.qdrant_collection, 

103 } 

104 missing = [k for k, v in required.items() if not v] 

105 if missing: 

106 raise ValueError(f"Missing required production environment variables: {', '.join(missing)}") 

107 if self.allow_dev_rbac: 

108 raise ValueError("ALLOW_DEV_RBAC must be false in production") 

109 return self
