38class ActiveDirectoryConnector(BaseEnterpriseConnector): 

39 """ 

40 Active Directory connector via LDAPS (LDAP over TLS). 

41 Real implementation — connects to AD using ldap3 library. 

42 

43 Authentication: Service account credentials from Vault (not env vars). 

44 Connection: LDAPS (port 636) with TLS certificate validation. 

45 Operations: User sync, group sync, org unit queries (read-only). 

46 """ 

47 key = "active_directory" 

48 name = "Active Directory" 

49 system_type = "active_directory" 

50 category = "Identity" 

51 auth_type = "ldaps_service_account" 

52 env_prefix = "AD" 

53 capabilities = ["users_sync", "groups_sync", "org_units", "keycloak_role_mapping"] 

54 allowed_roles = ["hsaai_admin", "auditor"] 

55 

56 def fetch_data(self, query: dict[str, Any], context: ConnectorContext) -> ConnectorResult: 

57 """ 

58 Query Active Directory via LDAPS. 

59 Returns real user/group data, not placeholder metadata. 

60 """ 

61 permission = self.check_permissions(context, "read") 

62 if not permission.success: 

63 return permission 

64 

65 import time 

66 start = time.time() 

67 

68 try: 

69 from ldap3 import Server, Connection, ALL, SUBTREE 

70 from ldap3.core.exceptions import LDAPException 

71 

72 # Read connection config from env (credentials from Vault in prod) 

73 ad_host = os.getenv("AD_HOST", "") 

74 ad_port = int(os.getenv("AD_PORT", "636")) 

75 ad_bind_dn = os.getenv("AD_BIND_DN", "") 

76 # In production: ad_password comes from Vault 

77 # ad_password = vault_client.get_secret_value("secret/data/hsaai/ad", "password") 

78 ad_password = os.getenv("AD_PASSWORD", "") 

79 ad_base_dn = os.getenv("AD_BASE_DN", "") 

80 use_ssl = ad_port == 636 

81 

82 if not all([ad_host, ad_bind_dn, ad_password, ad_base_dn]): 

83 return ConnectorResult( 

84 self.key, "fetch_data", False, data=None, 

85 source=self.name, 

86 message="AD not configured. Set AD_HOST, AD_BIND_DN, AD_PASSWORD, AD_BASE_DN", 

87 latency_ms=int((time.time() - start) * 1000), 

88 read_only=True, 

89 ) 

90 

91 # Connect via LDAPS 

92 server = Server(ad_host, port=ad_port, use_ssl=use_ssl, get_info=ALL) 

93 conn = Connection( 

94 server, user=ad_bind_dn, password=ad_password, 

95 auto_bind=True, read_only=True, 

96 ) 

97 

98 # Determine query type 

99 query_type = query.get("type", "users") 

100 search_filter = "(objectClass=*)" 

101 

102 if query_type == "users": 

103 search_filter = "(&(objectClass=user)(objectCategory=person))" 

104 elif query_type == "groups": 

105 search_filter = "(objectClass=group)" 

106 elif query_type == "org_units": 

107 search_filter = "(objectClass=organizationalUnit)" 

108 

109 # Search 

110 conn.search( 

111 search_base=ad_base_dn, 

112 search_filter=search_filter, 

113 search_scope=SUBTREE, 

114 attributes=["cn", "mail", "memberOf", "department", "title", "sAMAccountName"], 

115 size_limit=query.get("limit", 100), 

116 ) 

117 

118 # Parse results 

119 entries = [] 

120 for entry in conn.entries: 

121 entries.append({ 

122 "dn": entry.entry_dn, 

123 "cn": str(entry.cn) if hasattr(entry, "cn") else "", 

124 "mail": str(entry.mail) if hasattr(entry, "mail") else "", 

125 "department": str(entry.department) if hasattr(entry, "department") else "", 

126 "title": str(entry.title) if hasattr(entry, "title") else "", 

127 "sam_account_name": str(entry.sAMAccountName) if hasattr(entry, "sAMAccountName") else "", 

128 }) 

129 

130 conn.unbind() 

131 

132 return ConnectorResult( 

133 self.key, "fetch_data", True, 

134 data={ 

135 "type": query_type, 

136 "count": len(entries), 

137 "entries": entries[:50], # Limit response size 

138 "source": "active_directory_ldaps", 

139 }, 

140 source=self.name, 

141 message=f"Retrieved {len(entries)} {query_type} from AD", 

142 latency_ms=int((time.time() - start) * 1000), 

143 read_only=True, 

144 ) 

145 

146 except ImportError: 

147 return ConnectorResult( 

148 self.key, "fetch_data", False, data=None, 

149 source=self.name, 

150 message="ldap3 library not installed. Run: pip install ldap3", 

151 latency_ms=int((time.time() - start) * 1000), 

152 read_only=True, 

153 ) 

154 except LDAPException as e: 

155 return ConnectorResult( 

156 self.key, "fetch_data", False, data=None, 

157 source=self.name, 

158 message=f"AD LDAP error: {str(e)[:200]}", 

159 latency_ms=int((time.time() - start) * 1000), 

160 read_only=True, 

161 ) 

162 except Exception as e: 

163 return ConnectorResult( 

164 self.key, "fetch_data", False, data=None, 

165 source=self.name, 

166 message=f"AD connection error: {str(e)[:200]}", 

167 latency_ms=int((time.time() - start) * 1000), 

168 read_only=True, 

169 )
