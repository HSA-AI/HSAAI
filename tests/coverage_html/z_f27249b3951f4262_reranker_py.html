1""" 

2HSAAI Re-Ranker (v4.0 — AI-IMPROVEMENTS) 

3======================================== 

4 

5Production re-ranking for the RAG engine. Extends the v3.0 

6BM25 + semantic + proximity fusion with: 

7 

8 1. **Cross-encoder re-ranking** — uses a cross-encoder model 

9 (sentence-transformers `cross-encoder/ms-marco-MiniLM-L-6-v2` 

10 by default) to compute query-aware relevance scores. Falls back 

11 to a BM25+semantic proxy when the cross-encoder is unavailable. 

12 2. **MMR (Maximal Marginal Relevance)** — promotes diversity by 

13 penalizing chunks that are too similar to already-selected ones. 

14 3. **Business-context boosting** — boosts chunks whose `department` 

15 or `document_type` matches the query's inferred department. 

16 4. **Recency boost** — newer documents score higher (configurable 

17 half-life, default 180 days). 

18 5. **Explanation scores** — every returned chunk carries an 

19 `explanation` dict breaking down its score by component so 

20 callers can show "why this result ranked #1" in the UI. 

21 

22Backward compatibility: the v3.0 public API (`bm25_scores`, 

23`normalize_scores`, `rerank`) is preserved. v4.0 only adds new 

24optional kwargs and an `explanation` field on each result. 

25 

26The module is import-safe: heavy ML deps (sentence-transformers, 

27numpy) are imported lazily inside the cross-encoder path so the 

28reranker can be used in lightweight unit tests without them. 

29"""

68def bm25_scores(query: str, docs: list[str], k1: float = 1.5, b: float = 0.75) -> list[float]: 

69 q_tokens = tokenize(query) 

70 if not docs or not q_tokens: 

71 return [0.0 for _ in docs] 

72 tokenized = [tokenize(d) for d in docs] 

73 avgdl = sum(len(t) for t in tokenized) / max(1, len(tokenized)) 

74 df: Counter[str] = Counter() 

75 for toks in tokenized: 

76 df.update(set(toks)) 

77 scores = [] 

78 for toks in tokenized: 

79 tf = Counter(toks) 

80 dl = len(toks) or 1 

81 score = 0.0 

82 for term in q_tokens: 

83 if not tf[term]: 

84 continue 

85 idf = math.log(1 + (len(docs) - df[term] + 0.5) / (df[term] + 0.5)) 

86 denom = tf[term] + k1 * (1 - b + b * dl / max(1, avgdl)) 

87 score += idf * (tf[term] * (k1 + 1)) / denom 

88 scores.append(float(score)) 

89 return scores

92def normalize_scores(values: list[float]) -> list[float]: 

93 if not values: 

94 return [] 

95 lo, hi = min(values), max(values) 

96 if hi - lo < 1e-9: 

97 return [0.0 for _ in values] 

98 return [(v - lo) / (hi - lo) for v in values]

199def _cross_encoder_scores(query: str, docs: list[str]) -> Optional[list[float]]: 

200 """Score (query, doc) pairs with the cross-encoder. Returns None 

201 if the cross-encoder is unavailable.""" 

202 model = _get_cross_encoder() 

203 if model is None or not docs: 

204 return None 

205 try: 

206 pairs = [(query, d) for d in docs] 

207 raw = model.predict(pairs) 

208 # Cross-encoder returns logits — squash with sigmoid to [0,1]. 

209 out = [] 

210 for x in raw: 

211 try: 

212 x = float(x) 

213 except Exception: 

214 x = 0.0 

215 out.append(1.0 / (1.0 + math.exp(-x)) if x > -50 else 0.0) 

216 return out 

217 except Exception as exc: 

218 logger.warning("Cross-encoder prediction failed: %s", exc) 

219 return None

227def _parse_timestamp(ts: Any) -> Optional[float]: 

228 """Parse a timestamp from a chunk payload into a Unix epoch float. 

229 

230 Accepts: int/float epoch, ISO-8601 string, dict with `created_at` 

231 or `ts` key. 

232 """ 

233 if ts is None: 

234 return None 

235 if isinstance(ts, (int, float)): 

236 return float(ts) 

237 if isinstance(ts, str): 

238 # Try ISO-8601 first. 

239 try: 

240 from datetime import datetime 

241 if ts.endswith("Z"): 

242 ts = ts[:-1] + "+00:00" 

243 return datetime.fromisoformat(ts).timestamp() 

244 except Exception: 

245 try: 

246 return float(ts) 

247 except Exception: 

248 return None 

249 if isinstance(ts, dict): 

250 for key in ("created_at", "ts", "timestamp", "uploaded_at"): 

251 if key in ts: 

252 return _parse_timestamp(ts[key]) 

253 return None

256def _recency_boost(ts: Any, *, half_life_days: float = 180.0, now: Optional[float] = None) -> float: 

257 """Return a recency boost in [0, 1]. 

258 

259 Documents dated `now` → 1.0. Documents older than `half_life_days` 

260 decay exponentially. Documents with no timestamp get 0.5 (neutral). 

261 """ 

262 epoch = _parse_timestamp(ts) 

263 if epoch is None: 

264 return 0.5 

265 now = now if now is not None else time.time() 

266 age_days = max(0.0, (now - epoch) / 86400.0) 

267 return math.pow(0.5, age_days / half_life_days)

275def _business_boost(chunk: dict[str, Any], inferred_dept: Optional[str]) -> tuple[float, str]: 

276 """Return (boost_0_to_1, reason). Boosts chunks whose department 

277 or document_type matches the inferred department or the document 

278 type set.""" 

279 boost = 0.0 

280 reason = "" 

281 chunk_dept = (chunk.get("department") or chunk.get("domain") or "").lower().strip() 

282 chunk_dtype = (chunk.get("document_type") or chunk.get("doc_type") or "").lower().strip() 

283 

284 if inferred_dept and chunk_dept and inferred_dept in chunk_dept: 

285 boost = max(boost, 0.15) 

286 reason = f"dept_match:{chunk_dept}" 

287 if chunk_dtype and chunk_dtype in _DOCUMENT_TYPES: 

288 boost = max(boost, 0.10) 

289 if reason: 

290 reason += f"+doctype:{chunk_dtype}" 

291 else: 

292 reason = f"doctype:{chunk_dtype}" 

293 return boost, reason

301def _mmr_rerank( 

302 candidates: list[dict[str, Any]], 

303 *, 

304 lambda_: float = 0.7, 

305 top_k: int = 10, 

306 similarity_fn=None, 

307) -> list[dict[str, Any]]: 

308 """Greedy MMR selection. 

309 

310 Args: 

311 candidates: List of dicts each with `rerank_score` and `text`. 

312 lambda_: Trade-off between relevance (1.0) and diversity (0.0). 

313 top_k: Number of items to select. 

314 

315 Returns: 

316 Re-ordered list of length ≤ top_k maximizing: 

317 lambda_ * rel(item) - (1 - lambda_) * max_sim(item, selected) 

318 """ 

319 if not candidates: 

320 return [] 

321 if len(candidates) <= 1: 

322 return list(candidates) 

323 

324 # Default similarity: token Jaccard (cheap, no deps). 

325 if similarity_fn is None: 

326 def similarity_fn(a: str, b: str) -> float: 

327 ta = set(tokenize(a)) 

328 tb = set(tokenize(b)) 

329 if not ta or not tb: 

330 return 0.0 

331 return len(ta & tb) / len(ta | tb) 

332 

333 selected: list[dict[str, Any]] = [] 

334 remaining = list(candidates) 

335 max_rel = max((c.get("rerank_score", 0.0) for c in remaining), default=0.0) 

336 if max_rel <= 0: 

337 max_rel = 1.0 

338 

339 while remaining and len(selected) < top_k: 

340 best_item = None 

341 best_score = -float("inf") 

342 for item in remaining: 

343 rel = item.get("rerank_score", 0.0) / max_rel 

344 if not selected: 

345 diversity = 0.0 

346 else: 

347 diversity = max( 

348 similarity_fn(item.get("text", ""), s.get("text", "")) 

349 for s in selected 

350 ) 

351 mmr = lambda_ * rel - (1 - lambda_) * diversity 

352 if mmr > best_score: 

353 best_score = mmr 

354 best_item = item 

355 if best_item is None: 

356 break 

357 selected.append(best_item) 

358 remaining.remove(best_item) 

359 

360 return selected

400def rerank( 

401 query: str, 

402 hits: list[dict], 

403 semantic_weight: float = 0.30, 

404 lexical_weight: float = 0.35, 

405 proximity_weight: float = 0.10, 

406 cross_encoder_weight: float = 0.15, 

407 business_weight: float = 0.05, 

408 recency_weight: float = 0.05, 

409 *, 

410 use_cross_encoder: bool = True, 

411 use_mmr: bool = True, 

412 mmr_lambda: float = 0.7, 

413 mmr_top_k: Optional[int] = None, 

414 recency_half_life_days: float = 180.0, 

415 return_explanations: bool = True, 

416) -> list[dict]: 

417 """Re-rank RAG hits with cross-encoder + MMR + business + recency. 

418 

419 FIX-31: Rebalanced default weights — lexical_weight is now the largest 

420 single component (0.35, up from 0.25) and semantic_weight reduced (0.30, 

421 down from 0.35). Rationale: in hybrid retrieval, a strong lexical match 

422 (exact keyword/phrase overlap) is a more reliable signal of relevance 

423 than a noisy semantic similarity score, especially for Arabic queries 

424 where embedding models are less mature. The previous blend let a 

425 high-semantic-score but lexically-unrelated result outrank a strong 

426 lexical match, which broke user expectations (see 

427 test_hybrid_reranker_promotes_lexical_match). The new blend ensures 

428 that when lexical_score is high and semantic_score is moderate, the 

429 lexical match wins — while still allowing semantic signals to break 

430 ties and handle synonym/morphology cases. 

431 

432 Backward compatibility: 

433 The v3.0 positional signature (query, hits, semantic_weight, 

434 lexical_weight, proximity_weight) still works. v4.0 splits the 

435 weight budget differently to make room for the new layers — 

436 callers passing only positional args get the v4.0 default blend. 

437 

438 New kwargs: 

439 use_cross_encoder: If True, attempt to load and use the 

440 cross-encoder. If unavailable, falls back to a BM25+ 

441 semantic proxy with the same weight slot. 

442 use_mmr: If True, apply MMR diversity re-ranking after scoring. 

443 mmr_lambda: Relevance/diversity trade-off (1.0 = pure 

444 relevance, 0.0 = pure diversity). 

445 mmr_top_k: Number of results to keep after MMR (default: 

446 len(hits)). 

447 recency_half_life_days: Recency boost half-life. Documents 

448 older than this decay exponentially. 

449 return_explanations: If True, attach an `explanation` dict to 

450 each result. 

451 

452 Returns: 

453 List of dicts, sorted by `rerank_score` descending (or by MMR 

454 order if `use_mmr=True`). Each dict carries all original fields 

455 plus `semantic_score`, `lexical_score`, `cross_encoder_score`, 

456 `business_boost`, `recency_boost`, `mmr_penalty`, 

457 `rerank_score`, and (optionally) `explanation`. 

458 """ 

459 if not hits: 

460 return [] 

461 

462 docs = [str(h.get("text", "")) for h in hits] 

463 bm25 = normalize_scores(bm25_scores(query, docs)) 

464 semantic = normalize_scores([ 

465 float(h.get("score", h.get("semantic_score", 0.0)) or 0.0) 

466 for h in hits 

467 ]) 

468 

469 # Cross-encoder scores (or proxy if unavailable). 

470 ce_scores: Optional[list[float]] = None 

471 if use_cross_encoder: 

472 ce_scores = _cross_encoder_scores(query, docs) 

473 if ce_scores is None: 

474 # Proxy: average of normalized BM25 and normalized semantic. 

475 ce_scores = [ 

476 0.5 * (bm25[i] if i < len(bm25) else 0.0) + 

477 0.5 * (semantic[i] if i < len(semantic) else 0.0) 

478 for i in range(len(hits)) 

479 ] 

480 ce_used = False 

481 else: 

482 ce_used = True 

483 ce_scores = normalize_scores(ce_scores) 

484 

485 q_terms = set(tokenize(query)) 

486 inferred_dept = infer_department(query) 

487 now = time.time() 

488 

489 output: list[dict] = [] 

490 for i, h in enumerate(hits): 

491 # Proximity (carry-over from v3.0). 

492 terms = tokenize(str(h.get("text", ""))) 

493 proximity = 0.0 

494 if q_terms and terms: 

495 positions = [idx for idx, token in enumerate(terms) if token in q_terms] 

496 if len(positions) >= 2: 

497 span = max(positions) - min(positions) + 1 

498 proximity = min(1.0, len(positions) / max(1, span)) 

499 else: 

500 proximity = len(positions) / max(1, len(q_terms)) 

501 

502 # Business-context boost. 

503 biz_boost, biz_reason = _business_boost(h, inferred_dept) 

504 

505 # Recency boost. 

506 recency = _recency_boost( 

507 h.get("created_at") or h.get("uploaded_at") or h.get("timestamp"), 

508 half_life_days=recency_half_life_days, 

509 now=now, 

510 ) 

511 

512 ce_i = ce_scores[i] if i < len(ce_scores) else 0.0 

513 sem_i = semantic[i] if i < len(semantic) else 0.0 

514 lex_i = bm25[i] if i < len(bm25) else 0.0 

515 

516 # Weighted fusion. Weights are expected to sum to ~1.0; if they 

517 # don't, we normalize at the end. 

518 raw_total = ( 

519 semantic_weight * sem_i 

520 + lexical_weight * lex_i 

521 + proximity_weight * proximity 

522 + cross_encoder_weight * ce_i 

523 + business_weight * biz_boost 

524 + recency_weight * recency 

525 ) 

526 weight_sum = ( 

527 semantic_weight + lexical_weight + proximity_weight 

528 + cross_encoder_weight + business_weight + recency_weight 

529 ) 

530 final = raw_total / max(weight_sum, 1e-9) 

531 

532 item = dict(h) 

533 item["semantic_score"] = float(sem_i) 

534 item["lexical_score"] = float(lex_i) 

535 item["cross_encoder_score"] = float(ce_i) 

536 item["cross_encoder_used"] = bool(ce_used) 

537 item["proximity_score"] = float(proximity) 

538 item["business_boost"] = float(biz_boost) 

539 item["recency_boost"] = float(recency) 

540 item["rerank_score"] = float(final) 

541 item["score"] = float(final) # backward-compat alias 

542 

543 if return_explanations: 

544 reason_parts = [] 

545 if ce_used: 

546 reason_parts.append("cross-encoder") 

547 else: 

548 reason_parts.append("bm25+semantic-proxy") 

549 if biz_reason: 

550 reason_parts.append(biz_reason) 

551 if recency > 0.6: 

552 reason_parts.append("recent") 

553 elif recency < 0.4: 

554 reason_parts.append("stale") 

555 item["explanation"] = RerankExplanation( 

556 cross_encoder_score=float(ce_i), 

557 semantic_score=float(sem_i), 

558 lexical_score=float(lex_i), 

559 proximity_score=float(proximity), 

560 business_boost=float(biz_boost), 

561 recency_boost=float(recency), 

562 mmr_penalty=0.0, # filled in by MMR step 

563 final_score=float(final), 

564 reason=", ".join(reason_parts), 

565 ).to_dict() 

566 output.append(item) 

567 

568 # Initial sort by rerank_score (MMR will reorder for diversity). 

569 output.sort(key=lambda x: x.get("rerank_score", x.get("score", 0.0)), reverse=True) 

570 

571 if use_mmr and len(output) > 1: 

572 top_k = mmr_top_k if mmr_top_k is not None else len(output) 

573 reordered = _mmr_rerank(output, lambda_=mmr_lambda, top_k=top_k) 

574 # Compute MMR penalty = (original_score - final_position_score). 

575 # We approximate the penalty as the drop in `rerank_score` from 

576 # the best candidate to the selected one. This is informative 

577 # for UI: "this result was selected to diversify the set, at a 

578 # small relevance cost of X". 

579 if reordered and return_explanations:
