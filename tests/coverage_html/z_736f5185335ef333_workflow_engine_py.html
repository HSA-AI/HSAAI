18 started = time.time() 

19 workflow_run_id = f"wf_{uuid.uuid4().hex[:12]}" 

20 steps = [s.model_dump() for s in req.steps] if req.steps else DEFAULT_STEPS 

21 trace=[] 

22 status = "completed_with_controls"

24 kind = step["type"] 

25 item = {"id": step["id"], "type": kind} 

26 if kind in {"agent", "llm"}: 

27 result = run_agent(AgentRunRequest(agent_id=step.get("input", {}).get("agent_id", "supervisor"), task=req.goal, context=req.context)) 

28 item.update(status=result["status"], agent_run_id=result["run_id"]) 

29 elif kind == "rag": 

30 result = unified_search(EnterpriseSearchRequest(query=req.goal, context=req.context, sources=["rag"])) 

31 item.update(status=result["status"], results_count=result["count"]) 

32 elif kind == "approval": 

33 item.update(status="waiting_approval", detail="Human approval required") 

34 elif kind == "policy_check": 

35 try: 

36 verified_scope(req.context.model_dump()) 

37 item.update(status="completed", detail="Scope validated; endpoint RBAC enforced by router") 

38 except ValueError: 

39 item.update(status="failed", detail="Explicit scope required") 

40 else: 

41 item.update(status="failed", detail="Integration executor not configured") 

42 trace.append(item) 

43 if item["status"] != "completed": 

44 status = item["status"] 

45 break 

46 elapsed = int((time.time()-started)*1000) 

47 record_event(ObservabilityEvent(event_type="workflow_run", component="workflow_engine", tenant_id=req.context.tenant_id, workspace_id=req.context.workspace_id, latency_ms=elapsed, success=status == "completed_with_controls", risk_level="medium", metadata={"workflow_id": req.workflow_id, "workflow_run_id": workflow_run_id})) 

48 return {"workflow_run_id": workflow_run_id, "workflow_id": req.workflow_id, "goal": req.goal, "status": status, "trace": trace, "elapsed_ms": elapsed}
