70async def _verify_with_keycloak_jwks_async(token: str) -> dict[str, Any]: 

71 """Async JWT verification using cached JWKS.""" 

72 options = {"verify_aud": VERIFY_KEYCLOAK_AUDIENCE, "require": ["exp", "iss", "aud", "sub"]} 

73 kwargs: dict[str, Any] = {"issuer": KEYCLOAK_ISSUER, "options": options} 

74 if VERIFY_KEYCLOAK_AUDIENCE: 

75 kwargs["audience"] = KEYCLOAK_AUDIENCE 

76 jwks = await _fetch_jwks_async() 

77 unverified_header = jwt.get_unverified_header(token) 

78 kid = unverified_header.get("kid") 

79 if not kid: 

80 raise JWTError("No kid in token header") 

81 key = next((k for k in jwks.get("keys", []) if k.get("kid") == kid), None) 

82 if not key: 

83 # Invalidate cache and retry once in case Keycloak rotated keys 

84 global _jwks_cache 

85 _jwks_cache = None 

86 jwks = await _fetch_jwks_async() 

87 key = next((k for k in jwks.get("keys", []) if k.get("kid") == kid), None) 

88 if not key: 

89 raise JWTError(f"No matching key for kid={kid}") 

90 return jwt.decode(token, key=jwt.PyJWK.from_dict(key).key, algorithms=["RS256"], **kwargs)

143async def verify_authorization(authorization: str | None) -> dict[str, Any]: 

144 """ 

145 Verify the bearer token against Keycloak OIDC. 

146 

147 FIX: Removed the ALLOW_DEV_RBAC bypass that previously allowed 

148 unauthenticated access with admin privileges. Every request must 

149 now present a valid JWT token signed by Keycloak. 

150 

151 SECURITY FIX v2.1 (P0): This function is async. All FastAPI dependency 

152 wrappers (require_permission, require_any_role, get_current_claims) are 

153 now also async and properly await this function. 

154 """ 

155 if not authorization: 

156 raise HTTPException(status_code=401, detail="Missing bearer token. Authentication is required.") 

157 if not authorization.lower().startswith("bearer "): 

158 raise HTTPException(status_code=401, detail="Invalid authorization header format. Expected: Bearer <token>") 

159 token = authorization.split(" ", 1)[1].strip() 

160 if not token: 

161 raise HTTPException(status_code=401, detail="Empty bearer token") 

162 

163 # FIX: No more bypass. Always verify the token. 

164 # FIX B-08: Use the async JWKS verification — was calling sync function that internally called async without await. 

165 try: 

166 claims = await _verify_with_keycloak_jwks_async(token) 

167 except (JWTError, httpx.HTTPError, TypeError, ValueError) as exc: 

168 # Keep compatibility with the existing auth_service verifier. 

169 try: 

170 claims = await _verify_with_auth_service(authorization) 

171 except (httpx.HTTPError, HTTPException) as http_exc: 

172 logger.warning("Token verification failed: JWKS error=%s, auth_service error=%s", exc, http_exc) 

173 raise HTTPException(status_code=401, detail="Invalid or expired bearer token") 

174 normalized = _normalize_claims(claims) 

175 try: 

176 tenant_id, workspace_id = verified_scope(claims) 

177 except ValueError as exc: 

178 raise HTTPException(status_code=403, detail=str(exc)) from exc 

179 tenant_context.set(tenant_id) 

180 workspace_context.set(workspace_id) 

181 authorization_context.set(authorization) 

182 return normalized
